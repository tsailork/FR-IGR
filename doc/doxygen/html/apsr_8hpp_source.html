66inline void get_thermodynamics_apsr_2d(const double U[4],
67 double S_val,
68 const Parameters& p,
69 double& P_phys,
70 double& P_phan,
71 double& P_reg,
72 double& a_reg,
73 double n_x = 1.0,
74 double n_y = 0.0) {
75 (void)n_x; (void)n_y;
76 double rho = std::max(p.POS_LIMITER_EPS, U[0]);
77 double u = U[1] / rho;
78 double v = U[2] / rho;
79 double E = U[3];
80
81 double e_kin = 0.5 * rho * (u * u + v * v);
82 P_phys = std::max(p.POS_LIMITER_EPS, (p.GAMMA - 1.0) * (E - e_kin));
83 P_phan = S_val / rho;
84
85 double theta = p.PPR_CONSTANT_THETA;
86 double P_diff = P_phys - P_phan;
87
88 if (p.PPR_USE_LIMITER) {
89 double p_diff_min = -p.PPR_C_POS * P_phys;
90 double p_diff_max = p.PPR_C_MAX * P_phys;
91 P_diff = std::clamp(P_diff, p_diff_min, p_diff_max);
92 P_phan = P_phys - P_diff;
93 }
94
95 P_reg = P_phys + theta * P_diff;
96 P_reg = std::max(p.POS_LIMITER_EPS, P_reg);
97
98 // Whitham non-equilibrium sound speed
99 double a2_reg = ((1.0 + std::max(0.0, theta)) / rho) * (P_phys + (p.GAMMA - 1.0) * P_reg);
100 a_reg = std::sqrt(std::max(1e-12, a2_reg));
101}
