21inline void get_thermodynamics(double rho, double rhou, double rhov, double E, double S,
22 double theta, double gamma, double eps,
23 double& P_phys, double& P_phan, double& P_reg, double& a_reg)
24{
25 double u = rhou / rho;
26 double v = rhov / rho;
27 double e_kin = 0.5 * rho * (u * u + v * v);
28 P_phys = std::max(eps, (gamma - 1.0) * (E - e_kin));
29 P_phan = S / rho;
30 P_reg = std::max(eps, P_phys + theta * (P_phys - P_phan));
31
32 double a2_reg = ((1.0 + std::max(0.0, theta)) / rho) * (P_phys + (gamma - 1.0) * P_reg);
33 double a2_phys = gamma * P_phys / rho;
34 a_reg = std::sqrt(std::max(a2_reg, a2_phys));
35}

51inline void get_pointwise_regularization(const CellDim<2>& c, int iy, int ix,
52 const Basis& basis, const Parameters& p,
53 double press_phys, double& press_eff,
54 double& tau_xx, double& tau_xy, double& tau_yy)
55{
56 tau_xx = 0.0; tau_xy = 0.0; tau_yy = 0.0;
57 press_eff = press_phys;
58
59 if (!p.ENABLE_PPR) return;
60
61 double rho = std::max(p.POS_LIMITER_EPS, c.get_U(0, iy, ix, p.N_PTS));
62
63 if (p.ENABLE_APSR) {
64 double dP_dx = 0.0, dP_dy = 0.0;
65 for (int k = 0; k < p.N_PTS; ++k) {
66 double r_x = std::max(p.POS_LIMITER_EPS, c.get_U(0, iy, k, p.N_PTS));
67 double u_x = c.get_U(1, iy, k, p.N_PTS) / r_x;
68 double v_x = c.get_U(2, iy, k, p.N_PTS) / r_x;
69 double E_x = c.get_U(3, iy, k, p.N_PTS);
70 double p_x = std::max(p.POS_LIMITER_EPS, (p.GAMMA - 1.0) * (E_x - 0.5 * r_x * (u_x * u_x + v_x * v_x)));
71
72 double r_y = std::max(p.POS_LIMITER_EPS, c.get_U(0, k, ix, p.N_PTS));
73 double u_y = c.get_U(1, k, ix, p.N_PTS) / r_y;
74 double v_y = c.get_U(2, k, ix, p.N_PTS) / r_y;
75 double E_y = c.get_U(3, k, ix, p.N_PTS);
76 double p_y = std::max(p.POS_LIMITER_EPS, (p.GAMMA - 1.0) * (E_y - 0.5 * r_y * (u_y * u_y + v_y * v_y)));
77
78 dP_dx += basis.D[ix][k] * p_x;
79 dP_dy += basis.D[iy][k] * p_y;
80 }
81 dP_dx *= (2.0 / c.dx);
82 dP_dy *= (2.0 / c.dy);
83
84 double grad_mag = std::sqrt(dP_dx * dP_dx + dP_dy * dP_dy);
85 double n_x = 1.0, n_y = 0.0;
86 if (grad_mag > 1e-10) {
87 n_x = dP_dx / grad_mag;
88 n_y = dP_dy / grad_mag;
89 }
90
91 double P_phan = c.S_field[iy * p.N_PTS + ix] / rho;
92 double tau_mag = c.theta_avg * 0.5 * (press_phys - P_phan + std::sqrt((press_phys - P_phan)*(press_phys - P_phan) + p.POS_LIMITER_EPS * p.POS_LIMITER_EPS));
93 double alpha = p.APSR_ALPHA;
94
95 tau_xx = tau_mag * ((1.0 - alpha) * n_x * n_x + alpha);
96 tau_xy = tau_mag * (1.0 - alpha) * n_x * n_y;
97 tau_yy = tau_mag * ((1.0 - alpha) * n_y * n_y + alpha);
98 } else {
99 double P_phan = c.S_field[iy * p.N_PTS + ix] / rho;
100 double P_reg = press_phys + c.theta_avg * (press_phys - P_phan);
101 press_eff = std::max(p.POS_LIMITER_EPS, P_reg);
102 }
103}

105inline void get_thermodynamics_3d(double rho, double rhou, double rhov, double rhow, double E, double S,
106 double theta, double gamma, double eps,
107 double& P_phys, double& P_phan, double& P_reg, double& a_reg)
108{
109 double u = rhou / rho;
110 double v = rhov / rho;
111 double w = rhow / rho;
112 double e_kin = 0.5 * rho * (u * u + v * v + w * w);
113 P_phys = std::max(eps, (gamma - 1.0) * (E - e_kin));

115 P_reg = std::max(eps, P_phys + theta * (P_phys - P_phan));

118 double a2_phys = gamma * P_phys / rho;
119 a_reg = std::sqrt(std::max(a2_reg, a2_phys));

163inline void prepare_riemann_states(const double UL[4], const double UR[4],
164 double SL, double SR,
165 double thetaL, double thetaR, int dir,
166 const Parameters& p,
167 double pL, double pR,
168 double& pL_reg, double& pR_reg,
169 double& aL_reg, double& aR_reg)
170{
171 pL_reg = pL;
172 pR_reg = pR;
173 double rhoL = std::max(p.POS_LIMITER_EPS, UL[0]);
174 double rhoR = std::max(p.POS_LIMITER_EPS, UR[0]);
175 aL_reg = std::sqrt(p.GAMMA * pL / rhoL);
176 aR_reg = std::sqrt(p.GAMMA * pR / rhoR);
177
178 if (!p.ENABLE_PPR) return;
179
180 if (p.ENABLE_APSR) {
181 double P_physL = pL, P_physR = pR, P_phanL, P_phanR;
182 APSR::get_thermodynamics_apsr_2d(UL, SL, p, P_physL, P_phanL, pL_reg, aL_reg);
183 APSR::get_thermodynamics_apsr_2d(UR, SR, p, P_physR, P_phanR, pR_reg, aR_reg);
184 if (dir == 1) {
185 pL_reg = pL + p.APSR_ALPHA * (pL_reg - pL);
186 pR_reg = pR + p.APSR_ALPHA * (pR_reg - pR);
187 aL_reg = std::sqrt(p.GAMMA * pL_reg / rhoL);
188 aR_reg = std::sqrt(p.GAMMA * pR_reg / rhoR);
189 }
190 } else {
191 double theta_f = 0.5 * (thetaL + thetaR);
192 double P_phanL, P_phanR;
193 get_thermodynamics(rhoL, UL[1], UL[2], UL[3], SL, theta_f, p.GAMMA, p.POS_LIMITER_EPS, pL, P_phanL, pL_reg, aL_reg);
194 get_thermodynamics(rhoR, UR[1], UR[2], UR[3], SR, theta_f, p.GAMMA, p.POS_LIMITER_EPS, pR, P_phanR, pR_reg, aR_reg);
195 }
