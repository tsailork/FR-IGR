35 void build(const Solver& solver) {
36 npts = solver.p.N_PTS;
37 if (npts > MAX_LIM_PTS) return;
38 P = npts - 1;
39 const size_t num_cells = solver.cells.size();
40 C.assign(num_cells, std::vector<std::vector<std::vector<double>>>(4, std::vector<std::vector<double>>(npts, std::vector<double>(npts, 0.0))));
41
42 // 1. Build BB transformation matrices
43 for (int j = 0; j < npts; ++j) {
44 double t = 0.5 * (solver.basis.z[j] + 1.0);
45 for (int i = 0; i < npts; ++i) {
46 M_BB2Nodal[j][i] = bernstein_B(i, P, t);
47 }
48 }
49 invert_matrix(npts, M_BB2Nodal, M_Nodal2BB);
50
51 sum_w_sq = 0.0;
52 for (int i = 0; i < npts; ++i) {
53 for (int j = 0; j < npts; ++j) {
54 w_BB[i][j] = 0.0;
55 for (int my = 0; my < npts; ++my) {
56 for (int mx = 0; mx < npts; ++mx) {
57 double weight = 0.25 * solver.basis.w[my] * solver.basis.w[mx];
58 w_BB[i][j] += weight * M_BB2Nodal[my][j] * M_BB2Nodal[mx][i];
59 }
60 }
61 sum_w_sq += w_BB[i][j] * w_BB[i][j];
62 }
63 }
64
65 // 2. Transform nodal states to BB control points C[c_idx][v][j][i]
66 #pragma omp parallel for schedule(static)
67 for (size_t c_idx = 0; c_idx < num_cells; ++c_idx) {
68 Cell* c = solver.cells[c_idx];
69 for (int v = 0; v < 4; ++v) {
70 double tmp[MAX_LIM_PTS][MAX_LIM_PTS];
71 for (int iy = 0; iy < npts; ++iy) {
72 for (int i = 0; i < npts; ++i) {
73 tmp[iy][i] = 0.0;
74 for (int ix = 0; ix < npts; ++ix) {
75 tmp[iy][i] += M_Nodal2BB[i][ix] * c->get_U(v, iy, ix, npts);
76 }
77 }
78 }
79 for (int j = 0; j < npts; ++j) {
80 for (int i = 0; i < npts; ++i) {
81 C[c_idx][v][j][i] = 0.0;
82 for (int iy = 0; iy < npts; ++iy) {
83 C[c_idx][v][j][i] += M_Nodal2BB[j][iy] * tmp[iy][i];
84 }
85 }
86 }
87 }
88 }
89
90 // 3. Perform C0 boundary control point harmonization between conforming neighbor cells
91 auto C_orig = C;
92
93 #pragma omp parallel for schedule(static)
94 for (size_t c_idx = 0; c_idx < num_cells; ++c_idx) {
95 Cell* c = solver.cells[c_idx];
96 // Left face (i = 0) with neighbor 0
97 if (c->neighbors[0] && c->neighbors[0]->level == c->level) {
98 int n_idx = c->neighbors[0]->cell_index;
99 if (n_idx >= 0 && static_cast<size_t>(n_idx) < num_cells && solver.cells[n_idx] == c->neighbors[0]) {
100 for (int v = 0; v < 4; ++v) {
101 for (int j = 0; j < npts; ++j) {
102 double avg = 0.5 * (C_orig[c_idx][v][j][0] + C_orig[n_idx][v][j][P]);
103 C[c_idx][v][j][0] = avg;
104 }
105 }
106 }
107 }
108 // Right face (i = P) with neighbor 1
109 if (c->neighbors[1] && c->neighbors[1]->level == c->level) {
110 int n_idx = c->neighbors[1]->cell_index;
111 if (n_idx >= 0 && static_cast<size_t>(n_idx) < num_cells && solver.cells[n_idx] == c->neighbors[1]) {
112 for (int v = 0; v < 4; ++v) {
113 for (int j = 0; j < npts; ++j) {
114 double avg = 0.5 * (C_orig[c_idx][v][j][P] + C_orig[n_idx][v][j][0]);
115 C[c_idx][v][j][P] = avg;
116 }
117 }
118 }
119 }
120 // Bottom face (j = 0) with neighbor 2
121 if (c->neighbors[2] && c->neighbors[2]->level == c->level) {
122 int n_idx = c->neighbors[2]->cell_index;
123 if (n_idx >= 0 && static_cast<size_t>(n_idx) < num_cells && solver.cells[n_idx] == c->neighbors[2]) {
124 for (int v = 0; v < 4; ++v) {
125 for (int i = 0; i < npts; ++i) {
126 double avg = 0.5 * (C_orig[c_idx][v][0][i] + C_orig[n_idx][v][P][i]);
127 C[c_idx][v][0][i] = avg;
128 }
129 }
130 }
131 }
132 // Top face (j = P) with neighbor 3
133 if (c->neighbors[3] && c->neighbors[3]->level == c->level) {
134 int n_idx = c->neighbors[3]->cell_index;
135 if (n_idx >= 0 && static_cast<size_t>(n_idx) < num_cells && solver.cells[n_idx] == c->neighbors[3]) {
136 for (int v = 0; v < 4; ++v) {
137 for (int i = 0; i < npts; ++i) {
138 double avg = 0.5 * (C_orig[c_idx][v][P][i] + C_orig[n_idx][v][0][i]);
139 C[c_idx][v][P][i] = avg;
140 }
141 }
142 }
143 }
144 }
145
146 // 4. Apply exact L2 conservative projection to preserve cell averages
147 #pragma omp parallel for schedule(static)
148 for (size_t c_idx = 0; c_idx < num_cells; ++c_idx) {
149 Cell* c = solver.cells[c_idx];
150 double r_avg, ru_avg, rv_avg, E_avg;
151 Limiters::compute_cell_average(*c, solver.basis, r_avg, ru_avg, rv_avg, E_avg, npts);
152 double U_avg[4] = { r_avg, ru_avg, rv_avg, E_avg };
153
154 for (int v = 0; v < 4; ++v) {
155 double calc_avg = 0.0;
156 for (int j = 0; j < npts; ++j) {
157 for (int i = 0; i < npts; ++i) {
158 calc_avg += w_BB[i][j] * C[c_idx][v][j][i];
159 }
160 }
161 double delta = calc_avg - U_avg[v];
162 for (int j = 0; j < npts; ++j) {
163 for (int i = 0; i < npts; ++i) {
164 C[c_idx][v][j][i] -= (delta / sum_w_sq) * w_BB[i][j];
165 }
166 }
167 }
168 }
169 }

174 double eval(size_t c_idx, int v, double r, double s) const {
175 if (c_idx >= C.size()) return 0.0;
176 double tr = 0.5 * (r + 1.0);
177 double ts = 0.5 * (s + 1.0);
178 double val = 0.0;
179 for (int j = 0; j < npts; ++j) {
180 double B_j = bernstein_B(j, P, ts);
181 for (int i = 0; i < npts; ++i) {
182 double B_i = bernstein_B(i, P, tr);
183 val += C[c_idx][v][j][i] * B_i * B_j;
184 }
185 }
186 return val;
187 }
