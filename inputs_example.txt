ENTROPY_LIMITER_EPS = 1e-4

# LIMITER_STRATEGY: Limiter strategy selection for positivity and entropy limiters.
# Selects the local element reconstruction and attenuation mechanism used whenever a cell
# violates density/pressure positivity or minimum specific entropy thresholds.
# All strategies support 2D/3D conforming and quadtree non-conforming AMR grids, PPR phantom pressure,
# sub-grid face checking, and exact L2 cell average conservation.
# Options:
#   - ZHANG_SHU (Default):
#       Standard Zhang-Shu uniform scaling limiter. Scales the entire high-order solution polynomial
#       linearly towards the element cell average U_avg: U(x) <- theta * (U(x) - U_avg) + U_avg.
#       Computes theta = min(1, theta_rho, theta_p) to enforce min(rho, p) >= eps.
#       Literature Reference:
#         * Zhang, X., & Shu, C.-W. (2010). "On positivity-preserving high-order discontinuous Galerkin
#           schemes for compressible Euler equations on general meshes." J. Comput. Phys., 229(23), 8918-8934.
#         * Zhang, X., & Shu, C.-W. (2011). "Maximum-principle-satisfying and positivity-preserving
#           high-order schemes for conservation laws: Survey and new developments." Proc. R. Soc. A, 467, 2752-2776.
#
#   - BBCH:
#       Bernstein-Bézier Convex Hull limiter. Transforms nodal solution values into Bernstein-Bézier
#       control points C_{j,i} via matrix inversion. Exploits the convex hull property of Bernstein
#       polynomials (min C_{j,i} <= U(x) <= max C_{j,i}) to bound localized oscillations while applying
#       an L2 conservative projection onto the zero-mean subspace to preserve exact cell averages.
#       Literature References & Derivations:
#         * Dumont, A., & Loubère, R. (2013). "Bernstein-Bézier representation and positivity preservation
#           for high-order DG schemes." J. Comput. Phys., 245, 344-367.
#         * Dzanic, C., & Witherden, F. D. (2022). "Positivity-preserving modal limiters for high-order
#           discontinuous Galerkin methods." Comput. Fluids, 240, 105423.
#         * Lai, M. J., & Schumaker, L. L. (2007). Spline Functions on Triangulations. Cambridge Univ. Press.
#
#   - MODAL:
#       Scale-aware Legendre modal damping limiter. Projects the nodal state onto orthogonal 2D Legendre
#       polynomials P_k(x)P_l(y) and applies scale-dependent modal attenuation sigma(kx, ky; theta) = theta^((kx/P)^2 + (ky/P)^2).
#       Damps high-frequency modes (P, P) exponentially faster than low-degree modes, preserving physical
#       linear and quadratic gradients. Recovers the pure P0 cell average (U_avg) as an exact limit at theta = 0,
#       guaranteeing 100% positivity and entropy robustness equal or superior to Zhang-Shu.
#       Literature References & Derivations:
#         * Persson, P.-O., & Peraire, J. (2006). "Sub-grid shock-capturing for high-order discontinuous
#           Galerkin methods." AIAA Paper 2006-112.
#         * Dzanic, C., & Witherden, F. D. (2022). "Positivity-preserving modal limiters for high-order
#           discontinuous Galerkin methods." Comput. Fluids, 240, 105423.
#
#   - HERMITE:
#       Local Hermite patch subgrid limiter. Constructs a local subgrid Hermite interpolation patch matching
#       the cell average U_avg and interface states U_L, U_R, U_B, U_T derived from adjacent cell averages.
#       Uses zero-mean cubic Hermite basis functions h_L(z) = 0.25(1-z)^2(2+z)-0.5 to form a smooth profile
#       coupled to neighbor elements, applying L2 zero-mean projection and bisecting on theta in [0, 1]
#       towards the exact P0 cell average limit (theta = 0) for 100% positivity robustness.
#       Literature References & Derivations:
#         * Ciallella, A., & Torlo, D. (2023). "Arbitrary high-order sub-cell Hermite limiters for
#           discontinuous Galerkin methods." J. Sci. Comput., 95(1), 22.
#         * Vilar, F. (2019). "Cell-average based neural network and sub-cell Hermite limiters for
#           high-order DG schemes." Comput. Math. Appl., 78(9), 2991-3011.
LIMITER_STRATEGY = ZHANG_SHU

[PPR]
# ENABLE_PPR: Enable Hyperbolic Non-Equilibrium Phantom Pressure Relaxation (PPR) shock capturing.
ENABLE_PPR = false

# PPR_WALL_BC: Boundary condition mode for phantom pressure (S = rho * P_phan) at physical and SBM wall faces.
#   - EQUILIBRATED: Anti-symmetric pressure difference (P_diff,ghost = -P_diff,face).
#                   Forces zero average pressure difference at the wall interface (P_diff,avg = 0).
#                   Best for preserving un-regularized physical boundary layers in smooth flow. (Default)
#   - PHYSICAL_DIRICHLET: Zero ghost pressure difference (P_phan,ghost = P_phys,ghost, S_ghost = rho_ghost * P_phys,ghost).
#                         Imposes 50% boundary layer relaxation intensity at the wall interface.
#   - CONSTANT_DIFFERENCE: Constant pressure difference across wall boundary (P_diff,ghost = P_diff,face).
#                          Recommended for strong shock reflections off solid walls.
PPR_WALL_BC = EQUILIBRATED

# --- Core PPR Shock Indicator & Relaxation Parameters ---
# PPR_N_CELLS_SHOCK: Target shock transition width in cell units (N_cells,shock).
#                    Scales non-equilibrium dissipation theta_e quadratically: theta_target ~ (N_cells_shock * (N+1))^2.
#                    Default = 2.5 (spans shock transition over 2 to 3 cells).
PPR_N_CELLS_SHOCK = 2.5

# PPR_C_TAU: Baseline scaling coefficient for the analytical relaxation timescale tau.
#            tau = C_tau_eff * (h_eff / (N+1)) / (a_phys + ||u|| + 1e-12).
#            Dynamic Mach-Adaptive PPR automatically adapts C_tau_eff = min(C_tau_base, 0.90 / (theta_target + 1.0))
#            to hard-guarantee zero Doppler phase-lag growth (C_tau * (theta + 1) <= 0.90) at all Mach numbers.
#            Default = 0.25 (base baseline).
PPR_C_TAU = 0.25

# PPR_C_POS: Non-negativity bound factor (c_pos) for regularized pressure (P_reg >= c_pos * P_phys).
#            Determines the upper bound on phantom pressure: P_phan <= (1 + (1 - c_pos)/theta) * P_phys.
#            Default = 0.05.
PPR_C_POS = 0.05

# PPR_C_MAX: Maximum peak regularization factor (C_max) for regularized pressure (P_reg <= C_max * P_phys).
#            Determines the lower bound on phantom pressure: P_phan >= (1 - (C_max - 1)/theta) * P_phys.
#            Default = 2.5.
PPR_C_MAX = 2.5

# --- Advanced Multidimensional PPR Enhancements & Switches ---
# PPR_USE_SHOCK_NORMAL_MACH: Use shock-normal Mach number M_n = |u . grad(P)| / a in Master Law (default true).
#                             When true, projects fluid velocity onto local pressure gradient n_shock = grad(P)/|grad(P)|.
#                             Prevents tangential velocity along oblique shock flanks from over-inflating theta_e.
PPR_USE_SHOCK_NORMAL_MACH = true


# PPR_USE_DUCROS_SENSOR: Apply Ducros vorticity shielding to zero out PPR in shear layers and boundary layers (default true).
#                        Multiplies theta_e by Ducros ratio D_e = div(u)^2 / (div(u)^2 + curl(u)^2 + eps).
PPR_USE_DUCROS_SENSOR = true

# PPR_USE_ENERGY_GUARD: Apply Thermodynamic Instant-Thermalization Guard to zero out anti-dissipative states (default true).
#                       When true, forces P_phan -> P_phys in 1 RK stage if (P_phys - P_phan)*div(u) > 0 or P_phan > P_phys in compression.
PPR_USE_ENERGY_GUARD = true

# PPR_USE_STENCIL_EXPANSION: Expand theta_e and C_tau_cell to 1-ring face neighbors for smooth shock pre-conditioning (default true).
PPR_USE_STENCIL_EXPANSION = true

# PPR_USE_VON_NEUMANN_CEILING: Enforce conservative von Neumann ceiling C_tau_vonNeumann outside active shocks (default true).
#                              Prevents phase-lag ripple growth in smooth flow while bypassing ceiling inside active shocks.
PPR_USE_VON_NEUMANN_CEILING = true

# PPR_USE_SPATIAL_CLAMP: Clamp phantom pressure to local 3-element physical pressure min/max (default false).

PPR_USE_SPATIAL_CLAMP = false

# PPR_SENSOR_NOISE_FLOOR: Acoustic noise floor threshold for kinematic shock sensor (default 0.05).
PPR_SENSOR_NOISE_FLOOR = 0.05

PPR_CONSTANT_MODE = false

# PPR_CONSTANT_THETA: Coupling intensity theta used when PPR_CONSTANT_MODE = true (default 2.0).
PPR_CONSTANT_THETA = 2.0

# PPR_CONSTANT_C_TAU_VAL: Relaxation timescale ratio C_tau used when PPR_CONSTANT_MODE = true (default 0.25).
PPR_CONSTANT_C_TAU_VAL = 0.25

# --- Part A: Gradient-Guided Phantom Pressure Advection ---
# PPR_GRAD_ADV_SCALE: Acoustic pressure-gradient advection scale (default 0.0 = off).
# When > 0, corrects the phantom pressure advection velocity with an acoustic component
# oriented in the direction of the local pressure gradient:
#   u_adv = u + PPR_GRAD_ADV_SCALE * a * grad(P) / |grad(P)|
# The +sign pushes S toward the high-pressure (post-shock) side, evacuating it from
# the pre-shock region regardless of the sign of the flow velocity relative to the shock.
# PPR_GRAD_ADV_SCALE = 0.0 recovers classic advection; = 1.0 adds full acoustic magnitude.
# Values > 1.0 over-correct and may be useful experimentally.
PPR_GRAD_ADV_SCALE = 0.0

# PPR_GRAD_EPS: Regularization floor on |grad(P)| to prevent division-by-zero in
# smooth, uniform-pressure regions. Typically kept very small (1e-6 to 1e-8).
PPR_GRAD_EPS = 1e-6

# --- Part B: Adaptive & Smoothed Theta Shock Indicator ---
# In the updated PPR scheme, adaptive theta calculation and face-neighbor max-smoothing are
# built-in core defaults. The indicator theta_e is evaluated dynamically using:
#   1. Un-Limited Face Riemann Jump Sensor (100% P0-limiter immune): max(0, u_neigh - u_self) / ((N+1) * a_min)
#   2. Dynamic Mach-Adaptive Scaling: theta_target = ((N_cells_shock * (N+1))^2 / (pi^2 * C_tau_base)) * 0.5 * (gamma+1) * (1 + M_n^2) * shock_val
#   3. Phase-Lag Safe Relaxation: C_tau_eff = min(C_tau_base, 0.90 / (theta_target + 1.0)) guaranteeing C_tau * (theta + 1) <= 0.90
#   4. Ducros Vorticity Shielding & Optional Thermodynamic Energy Guard
#   5. Fast O(1) Pre-Computed Face-Neighbor Max Smoothing: theta_e = max(theta_raw, max_neighbor theta_raw)
PPR_DIV_ND_MAX = 2.0

# DOMAIN GRID SYNTAX (domain.grid)
################################################################################
# The simulation domain is defined in domain.grid using multiple [BlockN] 
# sections. Each block is a Cartesian grid with its own resolution and BCs.
#
# Constraints (due to 64-bit Morton spatial hashing):
#   - Maximum total number of blocks: 511
#   - Maximum block dimensions (N_ELEM_X, N_ELEM_Y): 2047 x 2047
#
# Example [Block0] configuration:
# [Block0]
# N_ELEM_X = 20           # Number of elements in X direction
# N_ELEM_Y = 20           # Number of elements in Y direction
# X_MIN = 0.0             # Physical X coordinate of the left face
# X_MAX = 1.0             # Physical X coordinate of the right face
# Y_MIN = 0.0             # Physical Y coordinate of the bottom face
# Y_MAX = 1.0             # Physical Y coordinate of the top face
# 
# BC_L = TRANSMISSIVE    # Boundary Condition for Left face
# BC_R = 1:L             # Connects to Block 1's Left face
# BC_B = 0:T             # Connects to Block 0's Top face (self-periodic)
# BC_T = 0:B             # Connects to Block 0's Bottom face (self-periodic)
#
# Available BCs:
#   - OUTFLOW_SUPERSONIC: Simple copy-out/ghost-state condition.
#   - WALL: Reflective slip-wall (normal velocity negated).
#   - WALL_NOSLIP: Adiabatic viscous no-slip wall (u=v=0, dT/dn=0).
#   - WALL_NOSLIP:T: Isothermal viscous no-slip wall at temperature T.
#   - WALL_MOVING:V: Adiabatic moving wall with tangential velocity V.
#   - WALL_MOVING:V:T: Isothermal moving wall with velocity V and temperature T.
#   - INFLOW_SUPERSONIC:rho:u:v:p: Enforces the given reference state.
#   - CHARACTERISTIC:rho:u:v:p: Riemann invariant based subsonic inflow/outflow using the reference state.
#   - TOTAL_PRESSURE_COMP:Pt: Subsonic backpressure boundary scaling the state to match target compressible total pressure Pt.
#   - TOTAL_PRESSURE_INCOMP:Pt: Subsonic backpressure boundary using Bernoulli's incompressible equation to match target total pressure Pt.
#   - STATIC_PRESSURE:P: Subsonic backpressure boundary enforcing the target static pressure P while copying interior density and velocity.
#   - ID:FACE: Connects to the specified face of neighbor Block ID.
#     Example: '2:T' means "Connect to the TOP face of Block 2".
#     Note: For periodic boundaries, simply map opposite faces to each other 
#     (e.g., Block 0 Left connects to Block 0 Right using '0:R').
#
#   Near-Wall Refinement Suffix Flags:
#     Any physical wall boundary (WALL, WALL_NOSLIP, WALL_MOVING) can optionally 
#     append a toggle suffix to control near-wall adaptive mesh refinement (AMR):
#       - :NOREFINED (or :NOREFINEMENT, :NO_REFINE) - disables near-wall refinement
#       - :REFINED (or :REFINE) - explicitly enables near-wall refinement (default)
#     Example: "BC_B = WALL:NOREFINED" ignores the bottom boundary during near-wall AMR.
